\section{Study Population}

\draftinstr{Follow the Phase 2 publication section
\texttt{final-protocol/publication/sections/sec-05-population.tex} and the Phase 1
model file \texttt{trial-protocol/draft-protocol/sections/sec-05-population.tex}.
Describe the population: adults with histologically confirmed, resectable or
borderline-resectable, KRAS G12-mutated PDAC, daraxonrasib-eligible under broad
pan-KRAS criteria \cite{rasolute302}, across eight HPB centers; $\sim 245$ screened
to yield 220 randomized 1:1 (110 per arm) allowing up to 10 percent
non-evaluability (\S9). Source from
\texttt{trial-protocol/nih-protocol/03\_study\_design\_and\_study\_population.md}.}

\draftinstr{Render \texttt{trial-phase-2/mermaid/fig-10-consort-flow.md} as a TikZ
\texttt{mermaidfig} (this is \textbf{Figure 10}): the randomized two-arm CONSORT
participant flow from $\sim 245$ assessed through exclusions, 220 randomized 1:1,
Arm A / Arm B ($\sim 110$ each), the discontinuation branch (\S7), and the ITT,
mITT, per-protocol, and safety analysis populations \cite{Schulz2010consort}.}

\subsection{Inclusion Criteria}
\draftinstr{Enumerate the inclusion criteria: signed consent with the Physical AI
opt-out election; age $\geq 18$; confirmed resectable / borderline-resectable PDAC;
documented KRAS G12 mutation; ECOG 0 or 1; adequate organ and marrow function;
measured baseline CA 19-9; daraxonrasib-eligible per RASolute 302 \cite{rasolute302};
and anatomy compatible with the eight-arm robotic system for those who may receive
Arm A. Include the synthetic exemplar PAT-PDAC-0002 (66-year-old, 3.2 cm head-of-
pancreas tumor abutting the SMV, KRAS G12D) after neoadjuvant modified FOLFIRINOX
\cite{Conroy2018}.}

\subsection{Exclusion Criteria}
\draftinstr{Enumerate the exclusion criteria: distant metastatic disease; vascular
involvement precluding R0; prior pancreatic resection precluding reconstruction;
uncontrolled intercurrent illness; pregnancy or lactation; contraindication or
hypersensitivity to daraxonrasib or modified FOLFIRINOX; and inability to consent
or comply. State equitable selection and the Patient Access and Equity Fund behind
the capital firewall \cite{kawchak2026hr9510,ecfr2026part54}.}

\subsection{Lifestyle Considerations}
\draftinstr{Perioperative nutrition plan, alcohol avoidance during dosing, tobacco
cessation support, routine-activity guidance, and the daraxonrasib labeling
interactions deferred to the concomitant-medication rules in \S6.}

\subsection{Screen Failures}
\draftinstr{Define screen failure and the minimal CONSORT screen-failure data set;
allow rescreening of transient disqualifiers with the same participant number, so
the screening-to-randomization accounting in Figure 10 stays complete.}

\subsection{Strategies for Recruitment and Retention}
\draftinstr{Recruitment across eight HPB centers at $\sim 10$ to 14 participants
per month over $\sim 30$ months; consent presented by a non-treating team member;
retention through structured follow-up and the Patient Access and Equity Fund
behind the capital firewall (travel, lodging, lost-wage, caregiver, navigation),
with OHRP-consistent safeguards and no undue inducement
\cite{kawchak2026hr9510,ecfr2026part54}.}
